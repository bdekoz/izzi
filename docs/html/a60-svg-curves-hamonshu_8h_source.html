39/**
40 * Deterministic procedural interpretations of the wave studies in Mori
41 * Yuzan's 1903 Hamonshu, volume 2.
42 *
43 * `pattern_specs` is the source-indexed catalogue. Pass any entry, a finite
44 * positive `pattern_box`, and an optional `motif_config` to
45 * `make_motif_path()` to obtain SVG path data suitable for
46 * `svg::make_path()`. The configuration changes line density, curvature,
47 * phase, orientation, reflection, and sampling without changing styling.
48 * Motifs use normalized coordinates, then scale independently in x and y to
49 * the requested box. Rotation may carry linework outside that box; callers
50 * that require a hard boundary should apply an SVG clip path.
51 *
52 * The routines construct only motif linework. Projection, geographic data,
53 * clipping, fill color, stroke style, and document/layer assembly remain the
54 * caller's responsibility. `curated_motif_selections` and
55 * `curated_curvature_ratios` define the shared 13-by-7 form set rendered by
56 * the parameter-grid example and cartofreako's ocean maps.
57 */

119struct pattern_box
120{
121 double left = std::numeric_limits<double>::infinity();
122 double top = std::numeric_limits<double>::infinity();
123 double right = -std::numeric_limits<double>::infinity();
124 double bottom = -std::numeric_limits<double>::infinity();
125
126 void
127 include(const svg::point_2t point)
128 {
129 const auto [x, y] = point;
130 left = std::min(left, x);
131 top = std::min(top, y);
132 right = std::max(right, x);
133 bottom = std::max(bottom, y);
134 }
135
136 bool
137 valid() const
138 {
139 return std::isfinite(left) && std::isfinite(top)
140 && std::isfinite(right) && std::isfinite(bottom)
141 && right > left && bottom > top;
142 }
143
144 double width() const { return right - left; }
145 double height() const { return bottom - top; }
146};

168/// Form and sampling parameters shared by every Hamonshu motif family.
169struct motif_config
170{
171 /// Multiplier for the number of repeated lines or constituent shapes.
172 double density = 1.0;
173 /// Multiplier for wave height, curl radius, and transverse displacement.
174 double curvature = 1.0;
175 /// Angular offset added to the catalogue motif's canonical phase, radians.
176 double phase = 0.0;
177 /// Rotation added to the catalogue motif's canonical orientation, radians.
178 double rotation = 0.0;
179 /// Reflect normalized motif coordinates around the box's vertical axis.
180 bool reflected = false;
181 /// Relative sampling resolution; 48 preserves the canonical resolution.
182 std::size_t samples_per_curve = 48;
183};

247inline void
248validate_pattern_spec(const pattern_spec& spec)
249{
250 require(spec.first_page >= 1 && spec.first_page <= spec.last_page
251 && spec.last_page <= 51,
252 "Hamonshu catalogue contains an invalid illustrated-page range");
253 require(!(spec.first_page <= 50 && spec.last_page >= 50),
254 "Hamonshu colophon page 50 cannot contain a wave motif");
255 require(spec.motif != 0 && !spec.name.empty(),
256 "Hamonshu catalogue contains an incomplete motif entry");
257 require(std::all_of(spec.name.begin(), spec.name.end(), [](const char value) {
258 const unsigned char character
259 = static_cast<unsigned char>(value);
260 return (character >= 'a' && character <= 'z')
261 || (character >= '0' && character <= '9')
262 || character == '-';
263 }),
264 "Hamonshu catalogue motif names must be lowercase ASCII slugs");
265}

267/// Resolve one stable source-coordinate selection into the full catalogue.
268inline const pattern_spec&
269curated_pattern(const curated_motif_selection selection)
270{
271 const auto match = std::find_if(
272 pattern_specs.begin(), pattern_specs.end(),
273 [=](const pattern_spec& spec) {
274 return spec.first_page == selection.first_page
275 && spec.last_page == selection.last_page
276 && spec.motif == selection.motif;
277 });
278 require(match != pattern_specs.end(),
279 "curated Hamonshu motif is not in the catalogue");
280 validate_pattern_spec(*match);
281 return *match;
282}

284inline bool
285has_token_prefix(const std::string_view text, const std::string_view prefix)
286{
287 std::size_t begin = 0;
288 while (begin < text.size())
289 {
290 const std::size_t end = text.find('-', begin);
291 const std::string_view token = text.substr(
292 begin, end == std::string_view::npos ? end : end - begin);
293 if (token.size() >= prefix.size()
294 && token.substr(0, prefix.size()) == prefix)
295 return true;
296 if (end == std::string_view::npos)
297 break;
298 begin = end + 1;
299 }
300 return false;
301}

368{

376inline constexpr std::size_t minimum_samples_per_curve = 8;
377inline constexpr std::size_t maximum_samples_per_curve = 4096;
378inline constexpr int maximum_repetition_count = 128;
379
380inline void
381validate_config(const motif_config& config)
382{
383 require(std::isfinite(config.density) && config.density >= 0.25
384 && config.density <= 4.0,
385 "Hamonshu density must be finite and between 0.25 and 4");
386 require(std::isfinite(config.curvature) && config.curvature >= 0.20
387 && config.curvature <= 4.0,
388 "Hamonshu curvature must be finite and between 0.20 and 4");
389 require(std::isfinite(config.phase),
390 "Hamonshu phase must be finite");
391 require(std::isfinite(config.rotation),
392 "Hamonshu rotation must be finite");
393 require(config.samples_per_curve >= minimum_samples_per_curve
394 && config.samples_per_curve <= maximum_samples_per_curve,
395 "Hamonshu samples_per_curve must be between 8 and 4096");
396}

398struct pattern_context
399{
400 pattern_box box;
401 unsigned key;
402 motif_config config;
403
404 int
405 repetition_count(const int canonical, const int minimum = 1) const
406 {
407 const double requested = canonical * config.density;
408 require(std::isfinite(requested)
409 && requested <= maximum_repetition_count,
410 "Hamonshu repetition count exceeds the supported limit");
411 return std::max(minimum, static_cast<int>(std::lround(requested)));
412 }
413
414 int
415 sample_count(const int canonical) const
416 {
417 const double requested
418 = canonical * static_cast<double>(config.samples_per_curve) / 48.0;
419 require(std::isfinite(requested)
420 && requested <= maximum_samples_per_curve,
421 "Hamonshu sample count exceeds the supported limit");
422 return std::max(
423 static_cast<int>(minimum_samples_per_curve),
424 static_cast<int>(std::lround(requested)));
425 }
426
427 svg::point_2t
428 point(double u, const double v) const
429 {
430 if (config.reflected)
431 u = 1 - u;
432 const double angle
433 = (static_cast<int>(key % 9) - 4) * (pi / 180)
434 + config.rotation;
435 const double du = u - 0.5;
436 const double dv = v - 0.5;
437 const double rotated_u = 0.5 + std::cos(angle) * du
438 - std::sin(angle) * dv;
439 const double rotated_v = 0.5 + std::sin(angle) * du
440 + std::cos(angle) * dv;
441 const svg::point_2t result {
442 box.left + rotated_u * box.width(),
443 box.top + rotated_v * box.height(),
444 };
445 const auto [x, y] = result;
446 require(std::isfinite(x) && std::isfinite(y),
447 "Hamonshu calculation produced a non-finite coordinate");
448 return result;
449 }
450};

452inline void
453append_polyline(std::string& path_data, const svg::vrange& points,
454 const bool close = false)
455{
456 if (points.size() < (close ? 3U : 2U))
457 return;
458 path_data += svg::make_path_data_from_points(points);
459 if (close)
460 path_data += "Z ";
461}

463template<typename Function>
464inline void
465append_curve(std::string& path_data, const pattern_context& context,
466 const int canonical_samples, Function function)
467{
468 const int samples = context.sample_count(canonical_samples);
469 svg::vrange points;
470 points.reserve(static_cast<std::size_t>(samples + 1));
471 for (int index = 0; index <= samples; ++index)
472 {
473 const double t = static_cast<double>(index) / samples;
474 const auto [u, v] = function(t);
475 points.push_back(context.point(u, v));
476 }
477 append_polyline(path_data, points);
478}

480inline void
481append_ellipse(std::string& path_data, const pattern_context& context,
482 const double center_u, const double center_v,
483 const double radius_u, const double radius_v,
484 const int canonical_samples = 28)
485{
486 const int samples = context.sample_count(canonical_samples);
487 svg::vrange points;
488 points.reserve(static_cast<std::size_t>(samples));
489 for (int index = 0; index != samples; ++index)
490 {
491 const double angle = 2 * pi * index / samples;
492 points.push_back(context.point(
493 center_u + radius_u * std::cos(angle),
494 center_v + radius_v * std::sin(angle)));
495 }
496 append_polyline(path_data, points, true);
497}

499inline void
500append_spiral(std::string& path_data, const pattern_context& context,
501 const double center_u, const double center_v,
502 const double radius, const double phase,
503 const double turns = 1.8)
504{
505 append_curve(
506 path_data, context, 42,
507 [=](const double t) {
508 const double angle = phase + turns * 2 * pi * t;
509 const double r = radius * (1 - 0.82 * t);
510 return std::pair {
511 center_u + r * std::cos(angle),
512 center_v + r * std::sin(angle),
513 };
514 });
515}

517inline void
518make_waterlines(std::string& data, const pattern_context& context)
519{
520 const int rows = context.repetition_count(
521 5 + static_cast<int>(context.key % 4));
522 const double phase = (context.key % 29) * 0.19
523 + context.config.phase;
524 for (int row = 0; row != rows; ++row)
525 append_curve(
526 data, context, 36,
527 [=](const double t) {
528 const double center = (row + 1.0) / (rows + 1.0);
529 const double wave = context.config.curvature * 0.035
530 * std::sin((2.0 + row % 3) * 2 * pi * t
531 + phase + row * 0.7);
532 const double ripple = context.config.curvature * 0.012
533 * std::sin(11 * pi * t + phase);
534 return std::pair {t, center + wave + ripple};
535 });
536}

538inline void
539make_crests(std::string& data, const pattern_context& context)
540{
541 const int rows = context.repetition_count(
542 3 + static_cast<int>(context.key % 2));
543 const int repeats = context.repetition_count(
544 3 + static_cast<int>((context.key / 3) % 3));
545 const double phase = context.config.phase / (2 * pi);
546 for (int row = 0; row != rows; ++row)
547 append_curve(
548 data, context, 48,
549 [=](const double t) {
550 double local = std::fmod(
551 t * repeats + 0.5 * (row % 2) + phase, 1.0);
552 if (local < 0)
553 local += 1;
554 const double arch = std::sin(pi * local);
555 const double center = (row + 1.0) / (rows + 1.0);
556 const double y = center
557 - context.config.curvature * 0.13 * arch * arch;
558 return std::pair {t, y};
559 });
560}

562inline void
563make_spirals(std::string& data, const pattern_context& context)
564{
565 const double phase = (context.key % 31) * 0.2
566 + context.config.phase;
567 const int count = context.repetition_count(
568 4 + static_cast<int>(context.key % 3));
569 const int rows = (count + 2) / 3;
570 const double radius = 0.16 * context.config.curvature
571 / std::sqrt(context.config.density);
572 for (int index = 0; index != count; ++index)
573 {
574 const double u = 0.18 + (index % 3) * 0.32;
575 const int row = index / 3;
576 const double v = rows <= 2
577 ? 0.28 + row * 0.42
578 : 0.16 + row * 0.68 / (rows - 1.0);
579 append_spiral(data, context, u, v, radius, phase + index * 0.8,
580 1.4 + 0.2 * (index % 3));
581 }
582}

584inline void
585make_spray(std::string& data, const pattern_context& context)
586{
587 const double phase = (context.key % 17) * 0.11
588 + context.config.phase;
589 const int branches = context.repetition_count(
590 5 + static_cast<int>(context.key % 3), 2);
591 for (int branch = 0; branch != branches; ++branch)
592 {
593 const double base = 0.1 + 0.8 * branch / (branches - 1.0);
594 append_curve(
595 data, context, 24,
596 [=](const double t) {
597 const double u = base + context.config.curvature * 0.10
598 * std::sin(pi * t + phase + branch);
599 const double v = 0.88 - 0.70 * t
600 + context.config.curvature * 0.08
601 * std::sin(3 * pi * t + branch + context.config.phase);
602 return std::pair {u, v};
603 });
604 append_ellipse(data, context,
605 base + 0.08 * std::sin(phase + branch),
606 0.12 + 0.03 * (branch % 2),
607 context.config.curvature * 0.018,
608 context.config.curvature * 0.025, 16);
609 }
610}

612inline void
613make_arcs(std::string& data, const pattern_context& context)
614{
615 const int rows = context.repetition_count(
616 4 + static_cast<int>(context.key % 2));
617 const int columns = context.repetition_count(5);
618 for (int row = 0; row != rows; ++row)
619 for (int column = -1; column <= columns; ++column)
620 append_curve(
621 data, context, 16,
622 [=](const double t) {
623 const double center = (column + 0.5 * (row % 2)) / columns;
624 const double angle = pi + pi * t + context.config.phase;
625 return std::pair {
626 center + context.config.curvature * 0.13 * std::cos(angle),
627 (row + 1.0) / (rows + 1.0)
628 + context.config.curvature * 0.10 * std::sin(angle),
629 };
630 });
631}

633inline void
634make_lattice(std::string& data, const pattern_context& context)
635{
636 const int lines = context.repetition_count(
637 7 + static_cast<int>(context.key % 3));
638 for (int line = -2; line < lines; ++line)
639 for (const double direction : {-1.0, 1.0})
640 append_curve(
641 data, context, 22,
642 [=](const double t) {
643 const double offset = line / static_cast<double>(lines);
644 const double u = t;
645 const double v = 0.5
646 + context.config.curvature * direction * (t - 0.5)
647 + offset - 0.35
648 + 0.05 * std::sin(context.config.phase);
649 return std::pair {u, v};
650 });
651}

653inline void
654make_bubbles(std::string& data, const pattern_context& context)
655{
656 const int count = context.repetition_count(
657 8 + static_cast<int>(context.key % 5));
658 for (int index = 0; index != count; ++index)
659 {
660 const unsigned mixed = context.key + static_cast<unsigned>(index * 97);
661 const double u = 0.12 + (mixed % 73) / 73.0 * 0.76;
662 const double v = 0.12 + ((mixed / 73) % 67) / 67.0 * 0.76;
663 const double radius = context.config.curvature
664 * (0.035 + ((mixed / 491) % 5) * 0.012)
665 / std::sqrt(context.config.density);
666 append_ellipse(data, context, u, v, radius,
667 radius * (0.8 + 0.12 * std::sin(context.config.phase)),
668 20);
669 }
670}

672inline void
673make_scrolls(std::string& data, const pattern_context& context)
674{
675 const int rows = context.repetition_count(
676 3 + static_cast<int>(context.key % 2));
677 const double phase = (context.key % 23) * 0.13
678 + context.config.phase;
679 for (int row = 0; row != rows; ++row)
680 {
681 append_curve(
682 data, context, 38,
683 [=](const double t) {
684 const double u = t;
685 const double v = (row + 1.0) / (rows + 1.0)
686 + context.config.curvature * 0.09
687 * std::sin(2 * pi * (2 * t) + phase);
688 return std::pair {u, v};
689 });
690 append_spiral(data, context,
691 0.18 + 0.64 * row / std::max(1, rows - 1),
692 (row + 1.0) / (rows + 1.0),
693 context.config.curvature * 0.10,
694 phase + row, 1.25);
695 }

698inline void
699make_fans(std::string& data, const pattern_context& context)
700{
701 const bool reverse = context.key % 2 != 0;
702 const int lines = context.repetition_count(8, 2);
703 for (int line = 0; line != lines; ++line)
704 append_curve(
705 data, context, 30,
706 [=](const double t) {
707 double u = 0.05 + 0.9 * t;
708 if (reverse)
709 u = 1 - u;
710 const double height = 0.10 + line * 0.315 / (lines - 1.0);
711 const double v = 0.88
712 - context.config.curvature * height
713 * std::sin(pi * t + context.config.phase)
714 - 0.28 * t;
715 return std::pair {u, v};
716 });
717}

719inline void
720make_breakers(std::string& data, const pattern_context& context)
721{
722 const bool reverse = context.key % 2 != 0;
723 const double phase = (context.key % 19) * 0.17
724 + context.config.phase;
725 const int lines = context.repetition_count(6, 2);
726 for (int line = 0; line != lines; ++line)
727 append_curve(
728 data, context, 38,
729 [=](const double t) {
730 double u = 0.04 + 0.78 * t;
731 if (reverse)
732 u = 1 - u;
733 const double v = 0.84
734 - context.config.curvature * 0.55
735 * std::sin(0.72 * pi * t)
736 + line * 0.125 / (lines - 1.0)
737 + context.config.curvature * 0.02 * std::sin(phase + 5 * t);
738 return std::pair {u, v};
739 });
740 append_spiral(data, context, reverse ? 0.22 : 0.78, 0.30,
741 context.config.curvature * 0.17,
742 (reverse ? pi : 0) + context.config.phase, 1.4);
743 const int drops = context.repetition_count(4, 2);
744 for (int drop = 0; drop != drops; ++drop)
745 append_ellipse(data, context,
746 0.42 + drop * 0.30 / (drops - 1.0),
747 0.16 + 0.04 * (drop % 2),
748 context.config.curvature * 0.012,
749 context.config.curvature * 0.018, 14);
750}

752inline void
753make_braids(std::string& data, const pattern_context& context)
754{
755 const int strands = context.repetition_count(
756 4 + static_cast<int>(context.key % 3), 2);
757 const double phase = (context.key % 13) * 0.23
758 + context.config.phase;
759 for (int strand = 0; strand != strands; ++strand)
760 append_curve(
761 data, context, 44,
762 [=](const double t) {
763 const double v = 0.5 + context.config.curvature * 0.22 * std::sin(
764 2 * pi * (1.5 + strand % 2) * t + phase
765 + strand * 2 * pi / strands);
766 return std::pair {t, v};
767 });
768}

770inline void
771make_cascade(std::string& data, const pattern_context& context)
772{

774 for (int line = 0; line != lines; ++line)
775 append_curve(
776 data, context, 32,
777 [=](const double t) {
778 const double u = 0.12 + line * 0.72 / (lines - 1.0)
779 + context.config.curvature * 0.07
780 * std::sin(pi * t + line * 0.4 + context.config.phase);
781 const double v = 0.05 + 0.9 * t;
782 return std::pair {u, v};
783 });

786inline void
787make_ripples(std::string& data, const pattern_context& context)
788{
789 const int centers = context.repetition_count(
790 2 + static_cast<int>(context.key % 2), 2);
791 const int rings = context.repetition_count(4);
792 for (int center = 0; center != centers; ++center)
793 for (int ring = 1; ring <= rings; ++ring)
794 append_ellipse(data, context,
795 0.28 + center * 0.43 / (centers - 1.0),
796 0.35 + 0.25 * (center % 2),
797 context.config.curvature * 0.035 * ring,
798 context.config.curvature * 0.022 * ring
799 * (1 + 0.12 * std::sin(context.config.phase)),
800 28);
801}

803inline void
804make_fountains(std::string& data, const pattern_context& context)
805{
806 const int lines = context.repetition_count(7, 2);
807 for (int line = 0; line != lines; ++line)
808 {
809 const double spread = context.config.curvature
810 * (0.12 + line * 0.33 / (lines - 1.0));
811 for (const double direction : {-1.0, 1.0})
812 append_curve(
813 data, context, 28,
814 [=](const double t) {
815 const double u = 0.5 + direction * spread * std::sin(pi * t / 2);
816 const double v = 0.92 - 0.78 * t
817 + context.config.curvature * 0.18 * t * t;
818 return std::pair {u, v};
819 });
820 }
821 append_spiral(data, context, 0.24, 0.72,
822 context.config.curvature * 0.10,
823 pi + context.config.phase, 1.1);
824 append_spiral(data, context, 0.76, 0.72,
825 context.config.curvature * 0.10,
826 context.config.phase, 1.1);
827}

829inline void
830make_clouds(std::string& data, const pattern_context& context)
831{
832 const double phase = (context.key % 31) * 0.1
833 + context.config.phase;
834 const int count = context.repetition_count(7);
835 const int rows = (count + 3) / 4;
836 for (int index = 0; index != count; ++index)
837 {
838 const double u = 0.12 + (index % 4) * 0.25;
839 const int row = index / 4;
840 const double v = rows <= 2
841 ? 0.30 + row * 0.38
842 : 0.18 + row * 0.64 / (rows - 1.0);
843 append_spiral(data, context, u, v,
844 context.config.curvature * 0.13
845 / std::sqrt(context.config.density),
846 phase + index * 0.7, 1.35);
847 }
848 make_waterlines(data, context);
849}

851inline void
852make_cells(std::string& data, const pattern_context& context)
853{
854 const int rows = context.repetition_count(4, 2);
855 const int columns = context.repetition_count(5, 2);
856 for (int row = 0; row != rows; ++row)
857 for (int column = 0; column != columns; ++column)
858 {
859 const double u = 0.10 + column * 0.80 / (columns - 1.0)
860 + 0.10 * (row % 2);
861 const double v = 0.14 + row * 0.72 / (rows - 1.0);
862 const double r = context.config.curvature
863 * (0.07 + 0.01 * ((context.key + row + column) % 3))
864 / std::sqrt(context.config.density);
865 append_ellipse(data, context, u, v, r, r * 0.75, 22);
866 }
