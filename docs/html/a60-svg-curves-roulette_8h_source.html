35/// Whether the moving circle rolls outside or inside the fixed circle.
36enum class roulette_kind
37{
38 epitrochoid, ///< Moving circle rolls around the fixed circle's exterior.
39 hypotrochoid, ///< Moving circle rolls around the fixed circle's interior.
40};

42/// Parameters for a circle rolling along a straight line.
43struct trochoid_config
44{
45 double rolling_radius = 1.0; ///< Radius `a` of the rolling circle.
46 double point_distance = 1.0; ///< Center-to-tracing-point distance `b`.
47 double turns = 4.0; ///< Number of rolling-circle revolutions to sample.
48 double phase = 0.0; ///< Initial tracing-point angle, in radians.
49 std::size_t samples_per_turn = 160; ///< Polyline samples per revolution.
50};

52/// Parameters for epi- and hypotrochoids with an automatically closed path.
53struct roulette_config
54{
55 /// Radius of the fixed circle, expressed as an integer ratio term.
56 std::size_t fixed_radius = 3;
57 /// Radius of the rolling circle, expressed as an integer ratio term.
58 std::size_t rolling_radius = 1;
59 /// Distance from the rolling-circle center to the tracing point.
60 double point_distance = 1.0;
61 /// Initial tracing-point angle relative to the rolling circle, in radians.
62 double phase = 0.0;
63 /// Polyline samples for each complete orbit of the rolling-circle center.
64 std::size_t samples_per_turn = 192;
65};

82inline void
83validate_output_transform(const svg::point_2t origin, const double scale)
84{
85 const auto [origin_x, origin_y] = origin;
86 require(std::isfinite(origin_x) && std::isfinite(origin_y),
87 "roulette origin must contain finite coordinates");
88 require(std::isfinite(scale) && scale > 0,
89 "roulette scale must be finite and positive");
90}

92inline std::size_t
93sample_count(const double turns, const std::size_t samples_per_turn)
94{
95 require(std::isfinite(turns) && turns > 0,
96 "roulette turn count must be finite and positive");
97 require(samples_per_turn >= minimum_samples_per_turn,
98 "roulette samples_per_turn must be at least eight");
99 const double requested = std::ceil(
100 turns * static_cast<double>(samples_per_turn));
101 require(std::isfinite(requested)
102 && requested <= static_cast<double>(maximum_sample_count),
103 "roulette sample count exceeds the supported limit");
104 return static_cast<std::size_t>(requested);
105}

107inline svg::point_2t
108to_svg_point(const svg::point_2t origin, const double scale,
109 const double x, const double y)
110{
111 const auto [origin_x, origin_y] = origin;
112 const svg::point_2t result {
113 origin_x + scale * x,
114 origin_y - scale * y,
115 };
116 const auto [result_x, result_y] = result;
117 require(std::isfinite(result_x) && std::isfinite(result_y),
118 "roulette calculation produced a non-finite coordinate");
119 return result;
120}

123path_from_points(const svg::vrange& points, const bool close)

125 require(points.size() >= 2,

135/// Return the number of fixed-circle orbits required to close a centered
136/// roulette whose two radius terms are integers.
137/// @param config Centered roulette parameters.
138/// @return Denominator of the reduced fixed-to-rolling radius ratio.
139/// @throws std::invalid_argument if either radius is zero.
140inline std::size_t
141roulette_completion_turns(const roulette_config& config)
142{
143 roulette_detail::require(config.fixed_radius != 0,
144 "roulette fixed_radius must be positive");
145 roulette_detail::require(config.rolling_radius != 0,
146 "roulette rolling_radius must be positive");
147 return config.rolling_radius
148 / std::gcd(config.fixed_radius, config.rolling_radius);
149}

151/// Generate a trochoid from a circle rolling along a straight line.
152///
153/// The canonical cycloid has `point_distance == rolling_radius`. A smaller
154/// distance produces a curtate trochoid; a larger distance produces a prolate
155/// trochoid. Cartesian y coordinates are reflected into SVG's downward y axis.
156/// @param origin SVG location corresponding to the formula's Cartesian origin.
157/// @param scale Multiplier applied to every generated coordinate.
158/// @param config Rolling circle, tracing point, duration, and sampling data.
159/// @return SVG path data containing one open polyline.
160/// @throws std::invalid_argument for non-finite or unsupported parameters.
161inline std::string
162make_trochoid_path(const svg::point_2t origin, const double scale,
163 const trochoid_config& config = {})
164{
165 roulette_detail::validate_output_transform(origin, scale);
166 roulette_detail::require(
167 std::isfinite(config.rolling_radius) && config.rolling_radius > 0,
168 "trochoid rolling_radius must be finite and positive");
169 roulette_detail::require(
170 std::isfinite(config.point_distance) && config.point_distance >= 0,
171 "trochoid point_distance must be finite and nonnegative");
172 roulette_detail::require(std::isfinite(config.phase),
173 "trochoid phase must be finite");
174
175 const std::size_t samples = roulette_detail::sample_count(
176 config.turns, config.samples_per_turn);
177 const double maximum_t = 2 * roulette_detail::pi * config.turns;
178 svg::vrange points;
179 points.reserve(samples + 1);
180 for (std::size_t index = 0; index <= samples; ++index)
181 {
182 const double t = maximum_t * static_cast<double>(index) / samples;
183 const double angle = t + config.phase;
184 const double x = config.rolling_radius * t
185 - config.point_distance * std::sin(angle);
186 const double y = config.rolling_radius
187 - config.point_distance * std::cos(angle);
188 points.push_back(roulette_detail::to_svg_point(origin, scale, x, y));
189 }
190 return roulette_detail::path_from_points(points, false);
191}

193/// Generate a cycloid, the common trochoid traced at the circle's edge.
194/// @param origin SVG location corresponding to the formula's Cartesian origin.
195/// @param scale Multiplier applied to every generated coordinate.
196/// @param config Trochoid settings; `point_distance` is replaced by the radius.
197/// @return SVG path data containing one open polyline.
198inline std::string
199make_cycloid_path(const svg::point_2t origin, const double scale,
200 trochoid_config config = {})
201{
202 config.point_distance = config.rolling_radius;
203 return make_trochoid_path(origin, scale, config);
204}

206/// Generate an automatically closed epi- or hypotrochoid.
207///
208/// Integer radius terms make the closure period exact: the path samples the
209/// denominator of their reduced ratio in complete turns, as described by the
210/// source article.
211/// @param origin Center of the fixed circle in SVG coordinates.
212/// @param scale Multiplier applied to the radius terms and point distance.
213/// @param kind Selects outside-circle or inside-circle rolling.
214/// @param config Radius ratio, tracing point, phase, and sampling data.
215/// @return Closed SVG path data.
216/// @throws std::invalid_argument for invalid geometry or sampling parameters.
217inline std::string
218make_roulette_path(const svg::point_2t origin, const double scale,
219 const roulette_kind kind,
220 const roulette_config& config = {})
221{
222 roulette_detail::validate_output_transform(origin, scale);
223 const std::size_t turns = roulette_completion_turns(config);
224 roulette_detail::require(
225 std::isfinite(config.point_distance) && config.point_distance >= 0,
226 "roulette point_distance must be finite and nonnegative");
227 roulette_detail::require(std::isfinite(config.phase),
228 "roulette phase must be finite");
229 if (kind == roulette_kind::hypotrochoid)
230 roulette_detail::require(
231 config.fixed_radius > config.rolling_radius,
232 "hypotrochoid fixed_radius must exceed rolling_radius");
233
234 const std::size_t samples = roulette_detail::sample_count(
235 static_cast<double>(turns), config.samples_per_turn);
236 const double fixed = static_cast<double>(config.fixed_radius);
237 const double rolling = static_cast<double>(config.rolling_radius);
238 const double signed_rolling = kind == roulette_kind::epitrochoid
239 ? rolling : -rolling;
240 const double center_radius = fixed + signed_rolling;
241 const double frequency = center_radius / rolling;
242 const double maximum_t = 2 * roulette_detail::pi * turns;
243
244 svg::vrange points;
245 points.reserve(samples + 1);
246 for (std::size_t index = 0; index <= samples; ++index)
247 {
248 const double t = maximum_t * static_cast<double>(index) / samples;
249 const double tracing_angle = frequency * t + config.phase;
250 double x = center_radius * std::cos(t);
251 double y = center_radius * std::sin(t);
252 if (kind == roulette_kind::epitrochoid)
253 {
254 x -= config.point_distance * std::cos(tracing_angle);
255 y -= config.point_distance * std::sin(tracing_angle);
256 }
257 else
258 {
259 x += config.point_distance * std::cos(tracing_angle);
260 y -= config.point_distance * std::sin(tracing_angle);
261 }
262 points.push_back(roulette_detail::to_svg_point(origin, scale, x, y));
263 }
264 return roulette_detail::path_from_points(points, true);
265}
